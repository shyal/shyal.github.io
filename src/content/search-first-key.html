 1def first_of_k(A, k, l=0, r=-1):
 2    if r == -1:
 3        r = len(A)-1
 4    m = (l + r) // 2
 5    if A[m] == k:
 6        if (m > 0 and A[m-1] != k) or m == 0:
 7            return m
 8        return first_of_k(A, k, l, r=m-1)
 9    elif A[m] < k:
10        return first_of_k(A, k, l=m+1, r=r)
11    else:
12        return first_of_k(A, k, l, r=m-1)
