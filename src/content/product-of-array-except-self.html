 1def productExceptSelf(nums):
 2    ans = [1] * len(nums)
 3    left, right = 1, 1
 4    for i in range(len(nums)):
 5        ans[i] *= left
 6        ans[~i] *= right
 7        left *= nums[i]
 8        right *= nums[~i]
 9    return ans
