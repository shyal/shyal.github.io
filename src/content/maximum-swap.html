 2    def maximumSwap(self, num):
 3        A = [*map(int, str(num))]
 4        last = {x: i for i, x in enumerate(A)}
 5        for i, x in enumerate(A):
 6            for d in range(9, x, -1):
 7                if last.get(d, 0) > i:

 9                    return int("".join(map(str, A)))
